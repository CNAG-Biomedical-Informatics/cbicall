export const GatkBadge = ({version}) => {
  const legacy = version === '3.5';
  return (
    <span
      className={`cbicallStackBadge ${legacy ? 'cbicallStackBadgeLegacy' : 'cbicallStackBadgeCurrent'}`}
      title={legacy ? 'Legacy GATK 3.5 workflow stack' : 'Current GATK 4.6 workflow stack'}>
      <span className="cbicallStackMark" aria-hidden="true">G</span>
      <span>GATK {version}</span>
    </span>
  );
};
